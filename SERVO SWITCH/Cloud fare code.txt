/* ---------- sound ---------- */
function ctx(){
  if(!AC){try{AC=new (window.AudioContext||window.webkitAudioContext)()}catch(e){}}

function tone(f,d,type,vol,when){
  if(muted)return;
  var c=ctx();if(!c||c.state!=='running')return;
  var o=c.createOscillator(),g=c.createGain(),t=c.currentTime+(when||0);

  g.gain.exponentialRampToValueAtTime(vol||0.15,t+0.01);

  o.connect(g);g.connect(c.destination);
  o.start(t);o.stop(t+d+0.02);
}

export default {
  async fetch(req, env) {
    try {
      if (!env.KV) return new Response("KV binding missing: add variable name KV in Settings > Bindings", { status: 500 });
      const url = new URL(req.url);

      if (url.pathname === "/api/poll") {
        if (url.searchParams.get("key") !== DEVICE_KEY)
          return new Response("no", { status: 401 });
        const raw = await env.KV.get("light_cmd");
        if (!raw) return new Response("0|none");
        const c = JSON.parse(raw);
        if (Date.now() - c.ts > 30000) return new Response(c.id + "|none");
        return new Response(c.id + "|" + c.action);
      }

      if (url.pathname === "/api/cmd" && req.method === "POST") {
        const b = await req.json().catch(() => ({}));
        if (!["on", "off"].includes(b.action)) return new Response("Bad action", { status: 400 });
        const raw = await env.KV.get("light_cmd");
        if (raw && Date.now() - JSON.parse(raw).ts < 3000)
          return new Response("Wait 3 seconds", { status: 429 });
        const now = Date.now();
        await env.KV.put("light_cmd", JSON.stringify({ id: String(now), action: b.action, ts: now }));
        return new Response(b.action === "on" ? "Light ON sent ✔" : "Light OFF sent ✔");
      }

      return new Response(PAGE, { headers: { "content-type": "text/html;charset=utf-8" } });
    } catch (e) {
      return new Response("Error: " + e.message, { status: 500 });
    }
  },
};
